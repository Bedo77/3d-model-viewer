    <script>
        // Debug logging function
        const debugEl = document.getElementById('debug');
        function log(message, type = 'info') {
            console[type === 'error' ? 'error' : 'log'](message);
            debugEl.innerHTML += `[${type.toUpperCase()}] ${new Date().toLocaleTimeString()}: ${message}<br>`;
            debugEl.scrollTop = debugEl.scrollHeight; // Auto-scroll
        }

        // Initialize when scene is ready
        const sceneEl = document.querySelector('a-scene');
        sceneEl.addEventListener('loaded', () => {
            log('A-Frame scene loaded successfully');
        });

        // Check AR.js initialization
        if (!window.THREEx || !window.THREEx.ArToolkitContext) {
            log('AR.js failed to load', 'error');
            throw new Error('AR.js not available');
        } else {
            log('AR.js initialized');
        }

        // Model loading function
        const loader = new THREE.GLTFLoader();
        function loadModel(entityId, url, scale = 1, positionOffset = { x: 0, y: 0, z: 0 }) {
            const entity = document.getElementById(entityId);
            if (!entity) {
                log(`Entity ${entityId} not found`, 'error');
                return;
            }

            log(`Loading model from ${url}`);
            loader.load(
                url,
                (gltf) => {
                    try {
                        // Adjust scale and position
                        gltf.scene.scale.set(scale, scale, scale);
                        gltf.scene.position.set(positionOffset.x, positionOffset.y, positionOffset.z);
                        
                        // Add to A-Frame entity
                        entity.object3D.add(gltf.scene);
                        log(`Model ${entityId} loaded successfully with scale ${scale}`);

                        // Bounding box for debugging
                        const box = new THREE.Box3().setFromObject(gltf.scene);
                        const size = new THREE.Vector3();
                        box.getSize(size);
                        log(`Model ${entityId} size: x:${size.x.toFixed(2)}, y:${size.y.toFixed(2)}, z:${size.z.toFixed(2)}`);
                    } catch (e) {
                        log(`Error processing ${entityId} model: ${e.message}`, 'error');
                    }
                },
                (xhr) => {
                    log(`${entityId}: ${(xhr.loaded / xhr.total * 100).toFixed(2)}% loaded`);
                },
                (error) => {
                    log(`Error loading ${entityId} from ${url}: ${error.message}`, 'error');
                }
            );
        }

        // Load models when marker is detected
        const marker = document.querySelector('a-marker');
        marker.addEventListener('markerFound', () => {
            log('Hiro marker detected');
            loadModel('insect', 
                'https://raw.githubusercontent.com/Bedo77/3d-model-viewer/main/models/Insect_Anatomy.glb', 
                2, // Scale up for visibility
                { x: 0, y: 0.5, z: 0 } // Slight Y offset
            );
            loadModel('bullet', 
                'https://raw.githubusercontent.com/Bedo77/3d-model-viewer/main/models/bullet_physics_animation_8_reconstruction.glb', 
                0.5, // Smaller scale
                { x: 0, y: 0, z: 0 }
            );
        });

        marker.addEventListener('markerLost', () => {
            log('Hiro marker lost');
        });

        // Webcam error handling
        navigator.mediaDevices.getUserMedia({ video: true })
            .then(() => log('Webcam access granted'))
            .catch((err) => log(`Webcam access denied: ${err.message}`, 'error'));

        // Animation loop for debugging
        function animate() {
            requestAnimationFrame(animate);
            const group = document.getElementById('model-group');
            if (group && marker.object3D.visible) {
                group.object3D.rotation.y += 0.01; // Slow rotation for visibility
            }
        }
        animate();

        // Initial log
        log('AR Test Script initialized');
    </script>

    // AR Mode function (WebXR example)
    function enterARMode() {
        if (!navigator.xr) {

        function loadModels(models, viewportId) {
            createScene(viewportId);
            const radius = 9;
            const angleIncrement = (Math.PI * 2) / models.length;

            models.forEach((model, index) => {
                const loader = new THREE.GLTFLoader();
                loader.load(model.path, function(object) {
                    if (object.scene) {
                        const angle = angleIncrement * index;
                        const x = Math.cos(angle) * radius;
                        const z = Math.sin(angle) * radius;

                        object.scene.position.set(x, 0, z);

                        switch (model.name) {
                            case 'Geometry Spheres':
                                object.scene.scale.set(2, 2, 2);
                                break;
                            case 'Insect Anatomy':
                                object.scene.scale.set(2, 2, 2);
                                object.scene.position.y = 0.5;
                                break;
                            case 'Female Human Skeleton':
                                object.scene.scale.set(0.105, 0.105, 0.105);
                                object.scene.position.y = 1.0;
                                break;
                            case 'Generalized Human Body':
                            case 'Animal Pack':
                                object.scene.scale.set(0.5, 0.5, 0.5);
                                break;
                            default:
                                object.scene.scale.set(model.scale.x, model.scale.y, model.scale.z);
                                break;
                        }

                        const spotLight = new THREE.SpotLight(model.color);
                        spotLight.position.set(x, 5, z);
                        spotLight.target = object.scene;
                        spotLight.castShadow = true;
                        currentScene.add(spotLight);
                        currentScene.add(spotLight.target);

                        modelGroup.add(object.scene);
                    }
                }, function(xhr) {
