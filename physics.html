function loadModels(models, viewportId) { createScene(viewportId); const radius = 9; // Radius for circular arrangement const angleIncrement = (Math.PI * 2) / models.length; // Even spacing models.forEach((model, index) => { console.log(`Loading model from: ${model.path}`); // Debugging statement const loader = new THREE.GLTFLoader(); loader.load(model.path, function(object) { if (object.scene) { // Position models in a circular arrangement const angle = angleIncrement * index; const x = Math.cos(angle) * radius; const z = Math.sin(angle) * radius; object.scene.position.set(x, 0, z); // Apply scaling and position adjustments based on model name switch (model.name) { case 'Lever and Pulley Set': object.scene.scale.set(2, 2, 2); // Double size break; case 'Force Simulator': object.scene.scale.set(2, 2, 2); // Double size object.scene.position.y = 0.5; // Lifted up on Y-axis by 0.5 units break; case 'EM Field Viewer': object.scene.scale.set(2, 2, 2); // Double size break; default: object.scene.scale.set(model.scale.x, model.scale.y, model.scale.z); // Default scaling break; } // Add a spotlight for each model with specific colors const spotLight = new THREE.SpotLight(model.color); spotLight.position.set(x, 5, z); // Height for better visibility spotLight.target = object.scene; // Aim at the model spotLight.castShadow = true; currentScene.add(spotLight); currentScene.add(spotLight.target); // Add target for spotlight modelGroup.add(object.scene); // Add the model to the model group } else { console.error(`Failed to load model: ${model.name}`); } }, function(xhr) { console.log((xhr.loaded / xhr.total * 100) + '% loaded'); }, function(error) { console.error(`Error loading model: ${model.path}`, error); alert(`Failed to load model: ${model.name}`); }); }); // Adjust camera position currentCamera.position.set(0, 10, 15); // Lifted the camera up by 2 degrees and brought it closer currentCamera.lookAt(0, 2, 0); // Adjusted focus point slightly higher }
